# ============================
# PLATFORM:
# LeetCode (Problem 1306 - Jump Game III)
# ============================

# ============================
# PROBLEM:
# You are given:
#
# - an array arr
# - a starting index start
#
# From index i, you can jump:
#
#     i + arr[i]
#     i - arr[i]
#
# Task:
# Return True if you can reach
# any index with value 0.
#
# Otherwise return False.
#
# Example:
# Input:
# arr = [4,2,3,0,3,1,2]
# start = 5
#
# Output:
# True
#
# Explanation:
# 5 -> 4 -> 1 -> 3
# arr[3] = 0
# ============================

# ============================
# APPROACH:
#
# Use BFS (Breadth First Search)
#
# Steps:
#
# 1. Start BFS from index = start.
#
# 2. For each index:
#
#    - If arr[index] == 0
#         return True
#
#    - Explore:
#         index + arr[index]
#         index - arr[index]
#
# 3. Use a visited set
#    to avoid revisiting indices.
#
# 4. If BFS ends without finding 0:
#       return False
#
# ============================

# ============================
# TIME COMPLEXITY:
# O(n)
#
# Each index visited once.
#
# SPACE COMPLEXITY:
# O(n)
# → queue + visited set
# ============================

from typing import List
import collections

class Solution:

    def canReach(self, arr: List[int], start: int) -> bool:

        # Queue for BFS
        q = collections.deque()

        q.append(start)

        # Visited indices
        visited = set()

        visited.add(start)

        # BFS traversal
        while q:

            indx = q.popleft()

            # Found zero
            if arr[indx] == 0:

                return True

            # Possible next jumps
            for x in (
                indx + arr[indx],
                indx - arr[indx]
            ):

                # Valid and not visited
                if (
                    0 <= x < len(arr)
                    and x not in visited
                ):

                    q.append(x)

                    visited.add(x)

        return False
